async function registerPasskey() { const btn = document.getElementById("passkeyBtn"); try { if (!window.isSecureContext) { alert("Face ID requires HTTPS."); return; } const { data: sessionData, error: sessionError } = await sb.auth.getSession(); if (sessionError || !sessionData?.session) { alert("Please log in first."); return; } if (!window.PublicKeyCredential) { alert("Passkeys are not supported in this browser."); return; } if (btn) { btn.disabled = true; btn.textContent = "SETTING UP FACE ID..."; } // Official Supabase Passkey registration const { data, error } = await sb.auth.registerPasskey(); if (error) { console.error("PASSKEY ERROR:", error); let msg = error.message || "Face ID setup failed."; if (error.code === "webauthn_credential_exists") { msg = "Face ID / Passkey is already registered for this account."; } alert(msg); return; } console.log("PASSKEY REGISTERED:", data); if (btn) { btn.textContent = "✓ FACE ID ACTIVE"; btn.disabled = true; } alert("Face ID is now active on this device."); } catch (err) { console.error("FACE ID ERROR:", err); if (err?.name === "NotAllowedError") { alert( "Face ID setup was cancelled or blocked by the iPhone browser. " + "Please try again directly from Safari." ); } else { alert( "Face ID could not be activated.\n\n" + (err?.message || err?.name || "Unknown error") ); } } finally { if (btn && btn.textContent === "SETTING UP FACE ID...") { btn.disabled = false; btn.textContent = "ACTIVATE FACE ID"; } } }
